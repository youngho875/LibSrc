cmake_minimum_required(VERSION 3.16)
project(TerrainTiles LANGUAGES CXX)
find_package(GDAL REQUIRED)
find_package(Threads REQUIRED)
add_library(TerrainTiles SHARED TerrainTiles.cpp)
target_compile_features(TerrainTiles PUBLIC cxx_std_17)
target_compile_definitions(TerrainTiles PRIVATE NOMINMAX)
target_link_libraries(TerrainTiles PRIVATE GDAL::GDAL Threads::Threads)
add_executable(test_tiles test_tiles.cpp)
target_compile_features(test_tiles PRIVATE cxx_std_17)
target_compile_definitions(test_tiles PRIVATE NOMINMAX)
target_link_libraries(test_tiles PRIVATE TerrainTiles GDAL::GDAL Threads::Threads)
