cmake_minimum_required(VERSION 3.20)
project(ModelAssets VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "C4I SDK root")
set(ASSIMP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/assimp-5.4.3" CACHE PATH "Assimp source")
set(ASSIMP_BINARY "${CMAKE_CURRENT_SOURCE_DIR}/third_party/build-assimp" CACHE PATH "Assimp build")
add_library(ModelAssets SHARED ModelAssets.cpp B3dm.cpp ModelImage.cpp ModelAssets.h ModelAssetsInternal.h)
target_compile_definitions(ModelAssets PRIVATE MODEL_ASSETS_EXPORTS NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(ModelAssets PRIVATE /utf-8 /W4)
target_include_directories(ModelAssets PRIVATE "${ASSIMP_ROOT}/include" "${ASSIMP_BINARY}/include" "${SDK_ROOT}/gdal_sdk/include")
# Link the exact configuration; never mix a Debug CRT library into Release.
file(GLOB ASSIMP_RELEASE "${ASSIMP_BINARY}/lib/Release/assimp*.lib")
file(GLOB DRACO_RELEASE "${ASSIMP_BINARY}/lib/Release/draco*.lib" "${ASSIMP_BINARY}/contrib/draco/Release/*.lib")
file(GLOB ZLIB_RELEASE "${ASSIMP_BINARY}/contrib/zlib/Release/*.lib")
if(NOT ASSIMP_RELEASE)
 message(FATAL_ERROR "Build Assimp Release first using build.ps1")
endif()
target_link_libraries(ModelAssets PRIVATE ${ASSIMP_RELEASE} ${DRACO_RELEASE} ${ZLIB_RELEASE} "${SDK_ROOT}/gdal_sdk/lib/gdal_i.lib" windowscodecs ole32)
set_target_properties(ModelAssets PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${SDK_ROOT}/LibSDK/bin" ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${SDK_ROOT}/LibSDK/lib")
add_custom_command(TARGET ModelAssets POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "${SDK_ROOT}/LibSDK/include/ModelAssets" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/ModelAssets.h" "${SDK_ROOT}/LibSDK/include/ModelAssets/ModelAssets.h")
