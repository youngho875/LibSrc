cmake_minimum_required(VERSION 3.16)
project(TerrainElevation LANGUAGES CXX)
find_package(GDAL REQUIRED)
find_package(Threads REQUIRED)
add_library(TerrainElevation SHARED TerrainElevation.cpp)
target_compile_features(TerrainElevation PUBLIC cxx_std_17)
target_include_directories(TerrainElevation PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(TerrainElevation PRIVATE GDAL::GDAL Threads::Threads)
if(MSVC)
 target_compile_options(TerrainElevation PRIVATE /utf-8 /W4)
 target_compile_definitions(TerrainElevation PRIVATE NOMINMAX)
endif()

include(CTest)
if(BUILD_TESTING)
 add_executable(test_terrain test_terrain.cpp)
 target_compile_features(test_terrain PRIVATE cxx_std_17)
 target_link_libraries(test_terrain PRIVATE TerrainElevation GDAL::GDAL Threads::Threads)
 add_test(NAME terrain_sdk COMMAND test_terrain)
endif()
